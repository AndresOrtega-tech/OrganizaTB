Script para crear la BD: -- 1. EXTENSIONES
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 2. TABLA DE PERFILES
CREATE TABLE public.profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    full_name TEXT,
    avatar_url TEXT,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. TABLA DE ETIQUETAS (Tags)
CREATE TABLE public.tags (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
    name TEXT NOT NULL,
    color TEXT DEFAULT '#808080',
    icon TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    -- Restricción: Un usuario no puede repetir nombre de tag, 
    -- pero diferentes usuarios sí pueden tener el mismo nombre.
    UNIQUE(user_id, name)
);

-- 4. TABLA DE TAREAS (Tasks)
CREATE TABLE public.tasks (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
    title TEXT NOT NULL,
    description TEXT,
    due_date TIMESTAMP WITH TIME ZONE,
    is_completed BOOLEAN DEFAULT FALSE,
    calendar_event_id TEXT, 
    has_reminder BOOLEAN DEFAULT FALSE,
    media_url TEXT NULL, -- Opcional para futuro soporte MCP
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. TABLA DE NOTAS (Notes)
CREATE TABLE public.notes (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
    title TEXT,
    content TEXT,
    is_archived BOOLEAN DEFAULT FALSE,
    media_url TEXT NULL, -- Opcional para futuro soporte MCP
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 6. TABLAS DE UNIÓN (Relación Muchos a Muchos con Tags)
CREATE TABLE public.task_tags (
    task_id UUID REFERENCES public.tasks(id) ON DELETE CASCADE,
    tag_id UUID REFERENCES public.tags(id) ON DELETE CASCADE,
    PRIMARY KEY (task_id, tag_id)
);

CREATE TABLE public.note_tags (
    note_id UUID REFERENCES public.notes(id) ON DELETE CASCADE,
    tag_id UUID REFERENCES public.tags(id) ON DELETE CASCADE,
    PRIMARY KEY (note_id, tag_id)
);

-- 7. TABLA DE ANÁLISIS (Improvement Insights)
CREATE TABLE public.improvement_insights (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL,
    type TEXT NOT NULL, 
    title TEXT NOT NULL,
    message TEXT,
    metrics JSONB DEFAULT '{}'::jsonb, 
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. SEGURIDAD (RLS)
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tags ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.tasks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.notes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.task_tags ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.note_tags ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.improvement_insights ENABLE ROW LEVEL SECURITY;

-- Políticas de Seguridad (RLS) Detalladas

-- 1. PROFILES
-- Permitir lectura y actualización del propio perfil. Insert se maneja por trigger.
DROP POLICY IF EXISTS "Privacidad" ON public.profiles;
CREATE POLICY "Profiles Select" ON public.profiles FOR SELECT USING (auth.uid() = id);
CREATE POLICY "Profiles Update" ON public.profiles FOR UPDATE USING (auth.uid() = id);

-- 2. TAGS
DROP POLICY IF EXISTS "Privacidad" ON public.tags;
-- Limpiar políticas anteriores si existen para evitar duplicados al recrear
DROP POLICY IF EXISTS "Tags Privacidad Select" ON public.tags;
DROP POLICY IF EXISTS "Tags Privacidad Update" ON public.tags;
DROP POLICY IF EXISTS "Tags Privacidad Delete" ON public.tags;
DROP POLICY IF EXISTS "Tags Privacidad Insert" ON public.tags;

CREATE POLICY "Tags Select" ON public.tags FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "Tags Update" ON public.tags FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Tags Delete" ON public.tags FOR DELETE USING (auth.uid() = user_id);
CREATE POLICY "Tags Insert" ON public.tags FOR INSERT WITH CHECK (auth.uid() = user_id);

-- 3. TASKS
DROP POLICY IF EXISTS "Privacidad" ON public.tasks;
CREATE POLICY "Tasks Select" ON public.tasks FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "Tasks Update" ON public.tasks FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Tasks Delete" ON public.tasks FOR DELETE USING (auth.uid() = user_id);
CREATE POLICY "Tasks Insert" ON public.tasks FOR INSERT WITH CHECK (auth.uid() = user_id);

-- 4. NOTES
DROP POLICY IF EXISTS "Privacidad" ON public.notes;
CREATE POLICY "Notes Select" ON public.notes FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "Notes Update" ON public.notes FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Notes Delete" ON public.notes FOR DELETE USING (auth.uid() = user_id);
CREATE POLICY "Notes Insert" ON public.notes FOR INSERT WITH CHECK (auth.uid() = user_id);

-- 5. IMPROVEMENT INSIGHTS
DROP POLICY IF EXISTS "Privacidad" ON public.improvement_insights;
CREATE POLICY "Insights Select" ON public.improvement_insights FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "Insights Update" ON public.improvement_insights FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Insights Delete" ON public.improvement_insights FOR DELETE USING (auth.uid() = user_id);
CREATE POLICY "Insights Insert" ON public.improvement_insights FOR INSERT WITH CHECK (auth.uid() = user_id);

-- 6. TABLAS DE UNIÓN (Task Tags y Note Tags)
-- Se valida la propiedad de la tarea/nota padre para permitir asociar tags.
DROP POLICY IF EXISTS "TaskTags Policy" ON public.task_tags;
CREATE POLICY "TaskTags Policy" ON public.task_tags FOR ALL USING (
    EXISTS (SELECT 1 FROM public.tasks WHERE id = task_tags.task_id AND user_id = auth.uid())
) WITH CHECK (
    EXISTS (SELECT 1 FROM public.tasks WHERE id = task_tags.task_id AND user_id = auth.uid())
);

DROP POLICY IF EXISTS "NoteTags Policy" ON public.note_tags;
CREATE POLICY "NoteTags Policy" ON public.note_tags FOR ALL USING (
    EXISTS (SELECT 1 FROM public.notes WHERE id = note_tags.note_id AND user_id = auth.uid())
) WITH CHECK (
    EXISTS (SELECT 1 FROM public.notes WHERE id = note_tags.note_id AND user_id = auth.uid())
);

-- 9. TRIGGER: CREACIÓN AUTOMÁTICA DE PERFIL
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.profiles (id, "Correo", full_name, avatar_url)
    VALUES (
        NEW.id, 
        NEW.email, 
        NEW.raw_user_meta_data->>'full_name', 
        NEW.raw_user_meta_data->>'avatar_url'
    );
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
    AFTER INSERT ON auth.users
    FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();
